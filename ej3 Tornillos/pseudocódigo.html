Algoritmo tornillos
    definir cod, contlote, cont, cant_fallas como entero;
    definir totalfalla, mesp, mtorni, error, limerror, porcfalote, min_fallo_lote, min_fallo_porc, may_fallo_lote, may_fallo_porc,maxerrabs como real;

    limerror ← 0.5;
    cont ← 0;

    Escribir 'Ingrese número de código (0 para terminar)';
    leer cod;

    min_fallo_porc ← 100; 
    may_fallo_porc ← 0;   

    Mientras cod ≠ 0 Hacer
        Escribir 'Ingrese medida esperada';
        leer mesp;
        
        cant_fallas ← 0;
        maxerrabs←0;
        
        Para contlote ← 1 Hasta 10 Hacer
            Escribir 'Ingrese medida del tornillo ',contlote;
            leer mtorni;
            error ← Abs(mesp - mtorni); 
        
            Si error > limerror Entonces
                cant_fallas ← cant_fallas + 1;
            FinSi
            
            si error>maxerrabs Entonces
                maxerrabs←error;
            FinSi
        FinPara
        
        porcfalote ← cant_fallas * 100 / contlote;
        totalfalla ← totalfalla + porcfalote;
        
        Escribir 'Porcentaje de fallas de este lote es: ', porcfalote;
        Escribir'El error máximo absoluto es: ',maxerrabs;
        
        Si porcfalote < min_fallo_porc Entonces
            min_fallo_porc ← porcfalote;
            min_fallo_lote ← cont + 1;
        FinSi
        
        Si porcfalote > may_fallo_porc Entonces
            may_fallo_porc ← porcfalote;
            may_fallo_lote ← cont + 1;
        FinSi
        
        cont ← cont + 1;
        
        Escribir 'Ingrese número de código (0 para terminar)';
        leer cod;
        FinMientras

    Si cont > 0 Entonces
        Escribir 'Cantidad de lotes procesados: ', cont;
        Escribir '% promedio de fallas: ', totalfalla/cont;
        Escribir 'Lote con menor cantidad de fallas: ', min_fallo_lote, ' con ', min_fallo_porc, '% de fallas';
        Escribir 'Lote con mayor cantidad de fallas: ', may_fallo_lote, ' con ', may_fallo_porc, '% de fallas';
    Sino
        Escribir 'No se procesaron lotes.';
    FinSi

finAlgoritmo
